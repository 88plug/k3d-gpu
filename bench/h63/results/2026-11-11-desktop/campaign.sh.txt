#!/usr/bin/env bash
# campaign.sh WHAT...: H63d on the desktop (i9-13900T). WHAT is smoke or a hold number (1-4), or
# "spare K J" to run spare hold K repeating hold J. Each step runs bench/h63/hold.sh desktop under
# ~/k3dgpu-bench.lock (hold.sh takes it; it also reads the co-tenants before and after). The
# campaign stops at the first step whose rc is not 0, so a spare is chosen by hand.
set -uo pipefail
W=/home/andrew/k3d-gpu/.claude/worktrees/wf_0b3703db-3b2-1
export G=/home/andrew/k3dgpu-h63 LOCK=/home/andrew/k3dgpu-bench.lock K3D_REAL=/usr/bin/k3d
export H63_IMAGE=cryptoandcoffee/k3d-gpu:0.1.105@sha256:22232fd444b345d40145a2931e756c0e50161cac7118b4ce8c5a724f2cfc495f
export SUDO_ASKPASS=/tmp/claude-1000/-home-andrew-k3d-gpu/86e13552-98b8-48e1-b1aa-9fbfea1a0978/scratchpad/h63d/askpass.sh
export PATH=${G}/bin:${PATH}
cd "${W}"
step() {  # step LABEL hold.sh-args...
    local label=$1 rc
    shift
    echo "$(date -u +%FT%TZ) ${label} start (commit $(git rev-parse --short HEAD), $(git status --porcelain | wc -l) uncommitted)" >> "${G}/campaign-steps.log"
    bash bench/h63/hold.sh desktop "$@"
    rc=$?
    echo "$(date -u +%FT%TZ) ${label} rc=${rc}" >> "${G}/campaign-steps.log"
    return "${rc}"
}
while [ $# -gt 0 ]; do
    case $1 in
        smoke) step smoke smoke || exit $? ; shift ;;
        spare) step "hold$2" "$2" "$3" || exit $? ; shift 3 ;;
        *) step "hold$1" "$1" || exit $? ; shift ;;
    esac
done
echo "$(date -u +%FT%TZ) done" >> "${G}/campaign-steps.log"
