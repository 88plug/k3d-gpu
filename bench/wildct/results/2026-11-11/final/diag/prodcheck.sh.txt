# read-only production checks on wildct; prints a stable, diffable block
echo "## containers (all): name state health restarts startedAt"
for c in $(timeout 30 docker ps -aq); do
  timeout 30 docker inspect -f '{{.Name}} {{.State.Status}} {{if .State.Health}}{{.State.Health.Status}}{{else}}-{{end}} restarts={{.RestartCount}} started={{.State.StartedAt}}' "$c"
done | sed 's#^/##' | sort
echo "## http (localhost)"
for u in http://127.0.0.1:16686/ http://127.0.0.1:9090/-/ready http://127.0.0.1:3100/ https://127.0.0.1:8443/ http://127.0.0.1:8890/healthz http://127.0.0.1:3080/ http://127.0.0.1:8081/ http://127.0.0.1:3000/; do
  printf '%s %s\n' "$u" "$(curl -sk -o /dev/null -w '%{http_code}' --max-time 15 "$u")"
done
echo "## prometheus up{} (job instance value)"
curl -s --max-time 15 'http://127.0.0.1:9090/api/v1/query?query=up' | python3 -c 'import json,sys; d=json.load(sys.stdin); [print(r["metric"].get("job"), r["metric"].get("instance"), r["value"][1]) for r in sorted(d["data"]["result"], key=lambda r:(r["metric"].get("job",""),r["metric"].get("instance","")))]' 2>&1
