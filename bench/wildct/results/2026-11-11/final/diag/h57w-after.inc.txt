# RESUME.md step 6's after-hold checks for an H57w hold (N set by the caller), read-only.
W=~/k3dgpu-bench-wildct; R=$W/repo; export PATH=$W/bin:$PATH
echo "== after $N $(date -u +%FT%TZ)"
cat $W/runs/$N/rc; ls $W/runs/$N/{flagged,disk-abort,cleanup-incomplete,unhealthy} 2>/dev/null; tail -3 $W/runs/campaign.log
ls $W/h57/results/H57w/hold*/{no-hold,flagged,disk-abort} 2>/dev/null
timeout 60 k3d cluster list -o json | python3 $R/bench/wildct/clusters.py; echo "list rc=$?"
timeout 30 docker ps -a --filter name=k3dgpu-w- --filter name=k3d-gpu-bench-load --format '{{.Names}}'; echo "ps rc=$?"
timeout 30 python3 $R/bench/wildct/strays.py $N; echo "strays rc=$?"
df -P -BG / | tail -1
flock -n $W/live.lock true && echo lock-free || echo lock-HELD
flock -n $W/inner.lock true && echo inner-free || echo inner-HELD
timeout 30 docker network ls --format '{{.Name}}' | grep -E 'k3dgpu' ; echo "net rc=$?"
timeout 30 docker volume ls --format '{{.Name}}' | grep -E 'k3dgpu' ; echo "vol rc=$?"
